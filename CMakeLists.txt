cmake_minimum_required(VERSION 3.16)
set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS: universal (Apple Silicon + Intel)")
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "macOS minimum version")
project(eps16emu C CXX)
if(APPLE)
  enable_language(OBJC)     # SDL .m files must be plain Objective-C (no -std=gnu++17)
  enable_language(OBJCXX)   # src/mac/menu_mac.mm (File / View menus)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(EPS16_GUI  "Build the SDL2 front end (eps16)" ON)
option(EPS16_MIDI "MIDI in/out through RtMidi" ON)
option(EPS16_ASIO "ASIO audio driver support on Windows (Steinberg ASIO SDK files, GPLv3)" ON)
if(APPLE)
  set(_bundled_sdl_default ON)
else()
  set(_bundled_sdl_default OFF)
endif()
option(EPS16_BUNDLED_SDL "Build SDL2 from source and link it statically (self-contained app)" ${_bundled_sdl_default})
set(SDL2_SOURCE_DIR "" CACHE PATH "Local SDL2 source tree (empty = download) for EPS16_BUNDLED_SDL")

# --- Musashi 68000 core
add_library(musashi STATIC
  musashi/m68kcpu.c musashi/m68kops.c musashi/m68kdasm.c musashi/softfloat/softfloat.c)
target_include_directories(musashi PUBLIC musashi musashi/softfloat src)

# --- machine core
add_library(eps16core STATIC
  src/eps16.cpp src/duart.cpp src/wd1772.cpp src/hd63450.cpp
  src/es5505.cpp src/es5510.cpp src/panel.cpp src/scsi.cpp src/efe_disk.cpp)
target_include_directories(eps16core PUBLIC src musashi)
target_link_libraries(eps16core PUBLIC musashi)

add_executable(eps16-cli src/cli.cpp)
target_link_libraries(eps16-cli eps16core)

if(EPS16_GUI)
  if(EPS16_BUNDLED_SDL)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
    if(SDL2_SOURCE_DIR STREQUAL "")
      include(FetchContent)
      FetchContent_Declare(sdl2 URL https://github.com/libsdl-org/SDL/releases/download/release-2.30.12/SDL2-2.30.12.tar.gz)
      FetchContent_MakeAvailable(sdl2)
    else()
      add_subdirectory(${SDL2_SOURCE_DIR} ${PROJECT_BINARY_DIR}/sdl2 EXCLUDE_FROM_ALL)
    endif()
  else()
    find_package(SDL2 REQUIRED)
  endif()
  add_executable(eps16 src/sdl_main.cpp src/vfd.cpp src/rack_ui.cpp src/ui_text.cpp src/native_menu.cpp src/audio_io.cpp src/settings.cpp third_party/rtaudio/RtAudio.cpp)
  if(APPLE)
    # EPS-16-Plus.app: ROMs and the OS disk in Contents/Resources/roms, icon, microphone permission
    set_target_properties(eps16 PROPERTIES
      MACOSX_BUNDLE TRUE
      OUTPUT_NAME "EPS-16-Plus"
      MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/src/mac/Info.plist.in
      MACOSX_BUNDLE_SHORT_VERSION_STRING "0.9"
      MACOSX_BUNDLE_BUNDLE_VERSION "0.9")
    target_sources(eps16 PRIVATE src/mac/eps16.icns src/mac/menu_mac.mm)
    target_link_libraries(eps16 "-framework Cocoa")
    set_source_files_properties(src/mac/eps16.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    file(GLOB _roms ${CMAKE_CURRENT_SOURCE_DIR}/roms/*.bin ${CMAKE_CURRENT_SOURCE_DIR}/roms/*.img)
    foreach(f ${_roms})
      target_sources(eps16 PRIVATE ${f})
      set_source_files_properties(${f} PROPERTIES MACOSX_PACKAGE_LOCATION Resources/roms)
    endforeach()
  endif()
  target_compile_definitions(eps16 PRIVATE ASR_WITH_RTAUDIO)
  target_include_directories(eps16 PRIVATE third_party/rtaudio/include)
  find_package(Threads REQUIRED)
  target_link_libraries(eps16 Threads::Threads)
  if(WIN32)
    target_compile_definitions(eps16 PRIVATE __WINDOWS_WASAPI__ __WINDOWS_DS__)
    if(MINGW)
      target_sources(eps16 PRIVATE src/win/wmcodecdsp_guids.c)
    else()
      target_link_libraries(eps16 wmcodecdsp)
    endif()
    target_link_libraries(eps16 ole32 winmm ksuser mfplat mfuuid dsound uuid comdlg32 shell32)
    if(EPS16_ASIO)
      target_compile_definitions(eps16 PRIVATE __WINDOWS_ASIO__)
      target_sources(eps16 PRIVATE third_party/rtaudio/include/asio.cpp third_party/rtaudio/include/asiodrivers.cpp
                                   third_party/rtaudio/include/asiolist.cpp third_party/rtaudio/include/iasiothiscallresolver.cpp)
    endif()
    target_sources(eps16 PRIVATE src/win/eps16.rc)
  elseif(APPLE)
    target_compile_definitions(eps16 PRIVATE __MACOSX_CORE__)
    target_link_libraries(eps16 "-framework CoreAudio" "-framework CoreFoundation" "-framework AudioToolbox")
  else()
    set(_rta_any OFF)
    find_package(ALSA)
    if(ALSA_FOUND)
      target_compile_definitions(eps16 PRIVATE __LINUX_ALSA__)
      target_link_libraries(eps16 ALSA::ALSA)
      set(_rta_any ON)
    endif()
    if(NOT _rta_any)
      target_compile_definitions(eps16 PRIVATE __RTAUDIO_DUMMY__)
    endif()
  endif()
  if(TARGET SDL2::SDL2-static)
    target_link_libraries(eps16 eps16core SDL2::SDL2-static)
  elseif(TARGET SDL2::SDL2)
    target_link_libraries(eps16 eps16core SDL2::SDL2)
  else()
    target_include_directories(eps16 PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(eps16 eps16core ${SDL2_LIBRARIES})
  endif()
  if(TARGET SDL2::SDL2main)
    target_link_libraries(eps16 SDL2::SDL2main)
  endif()
  if(EPS16_MIDI)
    target_sources(eps16 PRIVATE third_party/RtMidi.cpp)
    target_compile_definitions(eps16 PRIVATE EPS_WITH_RTMIDI)
    if(APPLE)
      target_compile_definitions(eps16 PRIVATE __MACOSX_CORE__)
      target_link_libraries(eps16 "-framework CoreMIDI" "-framework CoreAudio" "-framework CoreFoundation")
    elseif(WIN32)
      target_compile_definitions(eps16 PRIVATE __WINDOWS_MM__)
      target_link_libraries(eps16 winmm)
    elseif(ALSA_FOUND)
      target_compile_definitions(eps16 PRIVATE __LINUX_ALSA__)
    else()
      target_compile_definitions(eps16 PRIVATE __RTMIDI_DUMMY__)
    endif()
  endif()
endif()
